{{/* Explicit tabs computed from every category actually in use across the site's own content right now (every content page's category/categories frontmatter) — renders synchronously on mount instead of waiting on Pagefind's filters() to resolve, in a fixed, deliberate order instead of whatever order that API happens to enumerate in. Still entirely content-driven: a category with no pages simply never appears here, same as auto-derivation, just computed at build time instead of query time. */}} {{ $categories := slice }} {{ range site.RegularPages }} {{ with .Params.category }}{{ $categories = $categories | append . }}{{ end }} {{ range .Params.categories }}{{ $categories = $categories | append . }}{{ end }} {{ end }} {{ $categories = $categories | uniq | sort }} {{/* One default landing page per tab (config.tabs[].defaultPreviewUrl) — shown in the preview pane the moment a visitor switches to that tab with nothing typed yet, instead of the generic "type to search" empty state. Two passes: first, the earliest RegularPage in each category becomes the fallback (works for a category with no matching section, e.g. "Reference" -> api.md, "Resources" -> whichever of downloads/changelog sorts first); second, a category with a same-named section (e.g. "Guides" -> content/guides/_index.md) overrides that fallback with the section's own index page, a better landing page than any single guide. */}} {{ $categoryDefaults := dict }} {{ range site.RegularPages }} {{ $page := . }} {{ $pageCats := slice }} {{ with .Params.category }}{{ $pageCats = $pageCats | append . }}{{ end }} {{ range .Params.categories }}{{ $pageCats = $pageCats | append . }}{{ end }} {{ range $pageCats }} {{ if not (isset $categoryDefaults .) }} {{ $categoryDefaults = merge $categoryDefaults (dict . (dict "url" $page.RelPermalink "title" $page.Title)) }} {{ end }} {{ end }} {{ end }} {{ range $categories }} {{ $cat := . }} {{ with site.GetPage (printf "/%s" (lower $cat)) }} {{ $categoryDefaults = merge $categoryDefaults (dict $cat (dict "url" .RelPermalink "title" .Title)) }} {{ end }} {{ end }} {{/* Hugo-computed site-structure data (category list, per-category landing pages, and baseURL-correct exclude paths) that only the build itself can derive — everything else lives in docs/static/assets/js/ as plain external modules, loaded below in the same order this used to run inline. */}}
